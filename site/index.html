Quickstart · Open Python Referee CLI (eutropos-referee==1.0.1) 
pip install "git+https://github.com/different-change/eutropos.git@v1.0.1#subdirectory=referee"

# 1. Verify upstream Inspect Robots plugin entry points & hidden instruments
eutropos-referee prove-instruments --out-dir ./logs/a2

# 2. Seal a Protocol v1.0 run against live drand-quicknet & FreeTSA RFC 3161
eutropos-referee prove-v1 --live-crypto --out-dir ./logs/v1-live

# 3. Render CSP-hardened offline HTML audit report
eutropos-referee render-report --evidence-dir ./logs/v1-live/evidence_open \
  --public-key ./logs/v1-live/trusted-v1-open.pem \
  --require-live-beacon --require-live-tsa --out ./logs/v1-live/audit_report.html
